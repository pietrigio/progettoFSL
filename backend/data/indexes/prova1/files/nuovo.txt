from fastapi import FastAPI, WebSocket, UploadFile, File
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from typing import List
import os
import json
import asyncio
import shutil
import uuid

app = FastAPI()

# ===== CORS =====
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# ===== PATH =====
DATA_DIR = "data"
UPLOAD_DIR = "uploads"
INDEX_DIR = "indexes"
CHAT_FILE = os.path.join(DATA_DIR, "chats.json")

os.makedirs(DATA_DIR, exist_ok=True)
os.makedirs(UPLOAD_DIR, exist_ok=True)
os.makedirs(INDEX_DIR, exist_ok=True)

if not os.path.exists(CHAT_FILE):
    with open(CHAT_FILE, "w") as f:
        json.dump([], f)

# ===== MODELLI =====
class Message(BaseModel):
    role: str
    text: str

class Chat(BaseModel):
    id: int
    title: str
    messages: List[Message]

class IndexCreate(BaseModel):
    name: str

# ===== UTILS CHAT =====
def read_chats():
    with open(CHAT_FILE, "r") as f:
        return json.load(f)

def write_chats(chats):
    with open(CHAT_FILE, "w") as f:
        json.dump(chats, f, indent=2)

# ======================================================
# ===================== ENDPOINTS ======================
# ======================================================

# 🔹 ROOT
@app.get("/")
def root():
    return {"status": "FastAPI backend attivo"}

# 🔹 WEBSOCKET CHAT
@app.websocket("/ws")
async def websocket_chat(ws: WebSocket):
    await ws.accept()

    while True:
        data = await ws.receive_json()
        question = data.get("question", "")

        # --- RAG simulato ---
        response = f"Risposta simulata del modello alla domanda: {question}"

        for char in response:
            await ws.send_text(char)
            await asyncio.sleep(0.02)

        await ws.send_text("__END__")

# 🔹 CREA NUOVO INDEX
@app.post("/indexes/new")
def create_index(index: IndexCreate):
    path = os.path.join(INDEX_DIR, index.name)

    if not os.path.exists(path):
        os.makedirs(path)
        return {"ok": True, "message": "Index creato"}
    else:
        return {"ok": False, "message": "Index già esistente"}

# 🔹 LISTA INDEX
@app.get("/indexes")
def list_indexes():
    return [
        name for name in os.listdir(INDEX_DIR)
        if os.path.isdir(os.path.join(INDEX_DIR, name))
    ]

# 🔹 LISTA DOCUMENTI
@app.get("/documents")
def list_documents():
    return os.listdir(UPLOAD_DIR)

# 🔹 UPLOAD DOCUMENTO
@app.post("/upload")
def upload_document(file: UploadFile = File(...)):
    filename = f"{uuid.uuid4()}_{file.filename}"
    path = os.path.join(UPLOAD_DIR, filename)

    with open(path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)

    return {"ok": True, "filename": filename}

# 🔹 DELETE DOCUMENTO
@app.delete("/documents/delete")
def delete_document(filename: str):
    path = os.path.join(UPLOAD_DIR, filename)

    if os.path.exists(path):
        os.remove(path)
        return {"ok": True}
    else:
        return {"ok": False, "message": "File non trovato"}

# 🔹 CHAT (PERSISTENZA)
@app.get("/chats")
def get_chats():
    return read_chats()

@app.post("/chats")
def save_chat(chat: Chat):
    chats = read_chats()

    for i, c in enumerate(chats):
        if c["id"] == chat.id:
            chats[i] = chat.dict()
            break
    else:
        chats.insert(0, chat.dict())

    write_chats(chats)
    return {"ok": True}
